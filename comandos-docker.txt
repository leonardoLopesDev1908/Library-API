--> Para que dois ou mais containers consigam se comunicam é necessário criar uma network

# Criar network 
 docker network create library-network

# Rodar pgAdmin4
docker run --name pgadmin4 -p 15432:80 -e PGADMIN_DEFAULT_EMAIL=admin@admin.com -e PGADMIN_DEFAULT_PASSWORD=admin --network library-network dpage/pgadmin4

# Ver os processos rodando
docker ps

# Rodar psql db
docker run --name librarydb -p 5432:5432 -e POSTGRES_PASSWORD=postgres -e POSTGRES_USER=postgres -e POSTGRES_DB=library --network library-network postgres:17.5
